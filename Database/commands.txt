CREATING FIELDS/COLUMNS ON TABLE    

CREATE TABLE borrowers (
    borrower_id SERIAL PRIMARY KEY,
    borrower_no VARCHAR(20) NOT NULL UNIQUE,
    full_name VARCHAR(100) NOT NULL,
    borrower_type VARCHAR(20) NOT NULL CHECK (borrower_type IN ('Student', 'Faculty', 'Staff'))
);

CREATE TABLE equipment (
    equipment_id SERIAL PRIMARY KEY,
    property_no VARCHAR(30) NOT NULL UNIQUE,
    equipment_name VARCHAR(100) NOT NULL,
    quantity_available INT NOT NULL DEFAULT 0 CHECK (quantity_available >= 0)
);

CREATE TABLE borrow_transactions (
    borrow_id SERIAL PRIMARY KEY,
    borrower_id INT NOT NULL REFERENCES borrowers (borrower_id) ON DELETE RESTRICT,
    borrow_date DATE NOT NULL DEFAULT CURRENT_DATE,
    due_date DATE NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'Borrowed' CHECK (status IN ('Borrowed', 'Returned', 'Overdue', 'Cancelled')), 
    CHECK (due date >= borrow_date)
);

CREATE TABLE borrow_items (
    borrow_item_id SERIAL PRIMARY KEY,
    borrow_id INT NOT NULL REFERENCES borrow_transactions (borrow_id) ON DELETE CASCADE,
    equipment_id INT NOT NULL REFERENCES equipment (equipment_id) ON DELETE RESTRICT,
    quantity INT NOT NULL CHECK (quantity > 0),
    UNIQUE (borrow_id, equipment_id)
);

CREATE OR REPLACE FUNCTION fn_total_borrowed(equipment_id INT)
RETURNS INT AS $$
DECLARE
    total_qty INT;
BEGIN
    SELECT COALESCE(SUM(bi.quantity), 0) INTO total_qty
    FROM borrow_items bi
    JOIN borrow_transactions t ON bi.borrow_id = t.borrow_id
    WHERE bi.equipment_id = fn_total_borrowed.equipment_id AND t.status = 'Borrowed';
    
    RETURN total_qty;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION fn_check_equipment_availability(equipment_id INT, requested_quantity INT)
RETURNS BOOLEAN AS $$
DECLARE
    avail INT;
BEGIN
    SELECT quantity_available INTO avail FROM equipment WHERE equipment.equipment_id = fn_check_equipment_availability.equipment_id;
    IF FOUND AND avail >= requested_quantity THEN
        RETURN TRUE;
    END IF;
    RETURN FALSE;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION trg_validate_borrow_quantity_func()
RETURNS TRIGGER AS $$
BEGIN
    IF NOT fn_check_equipment_availability(NEW.equipment_id, NEW.quantity) THEN
        RAISE EXCEPTION 'Operation rejected: Requested quantity (%) exceeds available stock for equipment ID %', NEW.quantity, NEW.equipment_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_validate_borrow_quantity ON borrow_items;
CREATE TRIGGER trg_validate_borrow_quantity
BEFORE INSERT OR UPDATE ON borrow_items
FOR EACH ROW
EXECUTE FUNCTION trg_validate_borrow_quantity_func();


CREATE OR REPLACE FUNCTION trg_borrow_items_audit_func()
RETURNS TRIGGER AS $$
BEGIN
    IF (TG_OP = 'INSERT') THEN
        INSERT INTO borrow_items_audit (borrow_item_id, operation_type, new_borrow_id, new_equipment_id, new_quantity)
        VALUES (NEW.borrow_item_id, 'INSERT', NEW.borrow_id, NEW.equipment_id, NEW.quantity);
        RETURN NEW;
    ELSIF (TG_OP = 'UPDATE') THEN
        INSERT INTO borrow_items_audit (borrow_item_id, operation_type, old_borrow_id, old_equipment_id, old_quantity, new_borrow_id, new_equipment_id, new_quantity)
        VALUES (OLD.borrow_item_id, 'UPDATE', OLD.borrow_id, OLD.equipment_id, OLD.quantity, NEW.borrow_id, NEW.equipment_id, NEW.quantity);
        RETURN NEW;
    ELSIF (TG_OP = 'DELETE') THEN
        INSERT INTO borrow_items_audit (borrow_item_id, operation_type, old_borrow_id, old_equipment_id, old_quantity)
        VALUES (OLD.borrow_item_id, 'DELETE', OLD.borrow_id, OLD.equipment_id, OLD.quantity);
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_borrow_items_audit ON borrow_items;
CREATE TRIGGER trg_borrow_items_audit
AFTER INSERT OR UPDATE OR DELETE ON borrow_items
FOR EACH ROW
EXECUTE FUNCTION trg_borrow_items_audit_func();


CREATE TABLE borrow_items_audit (
    audit_id SERIAL PRIMARY KEY,
    borrow_item_id INT,
    operation_type VARCHAR(10),
    old_borrow_id INT,
    old_equipment_id INT,
    old_quantity INT,
    new_borrow_id INT,
    new_equipment_id INT,
    new_quantity INT,
    action_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
